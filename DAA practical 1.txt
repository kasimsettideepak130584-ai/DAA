#include <iostream>
#include <vector>
#include <cstdlib>
#include <ctime>
#include <chrono>
#include <algorithm>
#include <functional>

using namespace std;
using namespace std::chrono;

void bubbleSort(vector<int> &a) {
    for (int i = 0; i < a.size() - 1; ++i)
        for (int j = 0; j < a.size() - i - 1; ++j)
            if (a[j] > a[j + 1]) swap(a[j], a[j + 1]);
}

void selectionSort(vector<int> &a) {
    for (int i = 0; i < a.size() - 1; ++i) {
        int min = i;
        for (int j = i + 1; j < a.size(); ++j)
            if (a[j] < a[min]) min = j;
        swap(a[i], a[min]);
    }
}

void insertionSort(vector<int> &a) {
    for (int i = 1; i < a.size(); ++i) {
        int key = a[i], j = i - 1;
        while (j >= 0 && a[j] > key) { a[j + 1] = a[j]; j--; }
        a[j + 1] = key;
    }
}

void merge(vector<int> &a, int l, int m, int r) {
    vector<int> L(a.begin() + l, a.begin() + m + 1), R(a.begin() + m + 1, a.begin() + r + 1);
    int i = 0, j = 0, k = l;
    while (i < L.size() && j < R.size()) a[k++] = (L[i] <= R[j]) ? L[i++] : R[j++];
    while (i < L.size()) a[k++] = L[i++];
    while (j < R.size()) a[k++] = R[j++];
}

void mergeSort(vector<int> &a, int l, int r) {
    if (l < r) {
        int m = l + (r - l) / 2;
        mergeSort(a, l, m);
        mergeSort(a, m + 1, r);
        merge(a, l, m, r);
    }
}

int partition(vector<int> &a, int low, int high) {
    int pivot = a[high], i = low - 1;
    for (int j = low; j < high; ++j)
        if (a[j] < pivot) swap(a[++i], a[j]);
    swap(a[i + 1], a[high]);
    return i + 1;
}

void quickSort(vector<int> &a, int low, int high) {
    if (low < high) {
        int pi = partition(a, low, high);
        quickSort(a, low, pi - 1);
        quickSort(a, pi + 1, high);
    }
}

void benchmark(string label, function<void(vector<int>&)> sortFunc, vector<int> a) {
    auto start = high_resolution_clock::now();
    sortFunc(a);
    auto stop = high_resolution_clock::now();
    cout << label << " : " << duration_cast<microseconds>(stop - start).count() << " microseconds\n";
}

int main() {
    const int n = 100;
    vector<int> arr(n);
    srand(time(0));
    for (int &x : arr) x = rand() % 1000;

    cout << "Number of Elements = " << n << "\n\n";

    benchmark("Bubble Sort Time   ", bubbleSort, arr);
    benchmark("Selection Sort Time", selectionSort, arr);
    benchmark("Insertion Sort Time", insertionSort, arr);
    benchmark("Merge Sort Time    ", [n](vector<int> &v) { mergeSort(v, 0, n - 1); }, arr);
    benchmark("Quick Sort Time    ", [n](vector<int> &v) { quickSort(v, 0, n - 1); }, arr);

    return 0;
}
